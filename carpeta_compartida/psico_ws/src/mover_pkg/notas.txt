###### installs ######
sudo apt update
sudo apt install ros-noetic-teleop-twist-keyboard
sudo apt install ros-noetic-map-server
sudo apt install ros-noetic-move-base-msgs

###### guardar mapa ######
rosrun map_server map_saver -f path_maps_dir/nombre_mapa

###### lanzar mapa ######
rosrun map_server map_server path_maps_dir/nombre_mapa.yaml

###### lanzar teleop ######
source /opt/ros/noetic/setup.bash
rosrun teleop_twist_keyboard teleop_twist_keyboard.py cmd_vel:=mobile_base_controller/cmd_vel

###### mapear ######
mapping.launch en /home/robotica_gallium/share/tiago_2dnav/launch/mapping.launch
roslaunch tiago_2dnav mapping.launch
